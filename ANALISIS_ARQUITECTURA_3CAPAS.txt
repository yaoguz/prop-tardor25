// domaincontroller/CtrlDominioUsuario.java
public class CtrlDominioUsuario {
    private TreeMap<String, Usuario> usuarios;
    private GestorPersistencia gestorPersistencia;
    
    public CtrlDominioUsuario(GestorPersistencia gestor) {
        this.gestorPersistencia = gestor;
        this.usuarios = new TreeMap<>();
    }
    
    public String altaUsuario(String nomU, String contrasena, String confirmPass) {
        // Validaciones de negocio
        if (nomU == null || nomU.isEmpty()) 
            throw new IllegalArgumentException("Nombre vacio");
        if (!contrasena.equals(confirmPass))
            throw new IllegalArgumentException("Contrasenas no coinciden");
        if (usuarios.containsKey(nomU))
            throw new UsuarioYaExisteException("Usuario ya existe");
        
        // Crear usuario
        Usuario u = new Usuario(nomU, contrasena);
        usuarios.put(nomU, u);
        
        // Persistir (opcional, o al final del programa)
        gestorPersistencia.guardarUsuario(u);
        
        return nomU;
    }
    
    public boolean iniciarSesion(String nomU, String contrasena) {
        Usuario u = usuarios.get(nomU);
        if (u == null) throw new UsuarioNoExisteException();
        return u.checkContrasena(contrasena);
    }
    
    public Vector<String> consultaUsuarios() {
        Vector<String> resultado = new Vector<>();
        for (Usuario u : usuarios.values()) {
            resultado.add(u.getNombre());
        }
        return resultado;
    }
}
